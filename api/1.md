import json


# 最多向大模型展开多少条线程候选。调大可保留更多线索但增加 token；调小更精简但可能隐藏低排名候选。
MAX_TEMPLATE_CANDIDATES = 8
# 所有候选均未达阈值时最多展示多少条接近阈值的弱线索。调大可提高可解释性但增加 token；调小更精简。
MAX_TEMPLATE_DEVIATIONS = 3
# 每个方向最多展示多少个代表性异常值。调大可保留更多时间上下文但增加 token；调小更精简。
MAX_TEMPLATE_ANOMALY_POINTS = 3

DIRECTION_NAMES = {
    "cpu": "CPU",
    "net": "网络",
    "file": "文件操作",
    "epoll": "Epoll等待",
    "runq": "运行队列",
}
STATUS_NAMES = {
    "insufficient_data": "数据不足",
    "no_significant_anomaly": "未筛出达到阈值的候选",
    "potential_anomaly": "存在潜在异常",
    "strong_potential_anomaly": "存在强潜在异常",
}
TYPE_NAMES = {
    "potential_spike": "潜在线程尖峰",
    "spike": "强线程尖峰",
    "extreme_spike": "极端线程尖峰",
    "sparse_spike": "稀疏尖峰",
    "potential_step_change": "潜在阶跃突增",
    "step_change": "强阶跃突增",
    "activation_spike": "突然激活",
    "heavy_consumer": "重度消费者",
    "peer_outlier": "同方向离群",
    "collective_spike": "方向集体尖峰",
    "collective_activation": "方向集体激活",
    "collective_step_change": "方向集体阶跃",
}
LEVEL_NAMES = {
    "potential": "潜在",
    "strong": "强",
    "extreme": "极端",
}
TREND_NAMES = {
    "unknown": "未知",
    "stable": "平稳",
    "rising": "上升",
    "falling": "下降",
    "spike_then_fall": "先突增后回落",
    "rise_then_hold": "上升后维持",
    "intermittent": "间歇性活动",
}
PATTERN_NAMES = {
    "steady": "整体平稳",
    "isolated_spike": "孤立尖峰",
    "repeated_spikes": "多次尖峰",
    "potential_step": "潜在阶跃",
    "step_up": "阶跃上升",
    "activation": "突然激活",
    "sparse": "稀疏尖峰",
    "collective_spike": "方向集体尖峰",
    "collective_activation": "方向集体激活",
    "intermittent": "间歇性活动",
    "gradual_rise": "持续上升",
    "gradual_fall": "持续下降",
    "insufficient_data": "数据不足",
}
KIND_NAMES = {
    "peak": "峰值",
    "spike": "尖峰",
    "step": "阶跃",
    "activity": "活动",
    "collective_spike": "集体尖峰",
    "collective_activation": "集体激活",
    "collective_step": "阶跃",
}


def _format_value(value, unit):
    try:
        value = float(value)
    except (TypeError, ValueError):
        return "-"
    if unit == "ns":
        if abs(value) >= 1_000_000_000:
            return f"{value / 1_000_000_000:.3g}s"
        if abs(value) >= 1_000_000:
            return f"{value / 1_000_000:.3g}ms"
        if abs(value) >= 1_000:
            return f"{value / 1_000:.3g}us"
    return f"{value:.3g}{unit}"


def _format_ratio(value):
    return "-" if value is None else f"{float(value):.2f}x"


def _format_z(value):
    return "-" if value is None else f"{float(value):.2f}"


def _format_trend(summary, unit):
    if not summary:
        return "趋势未知"
    trend = _format_trend_name(summary)
    first = _format_value(summary.get("firstValue"), unit)
    last = _format_value(summary.get("lastValue"), unit)
    ratio = _format_ratio(summary.get("firstToLastRatio"))
    return f"{trend}；首段{first}；末段{last}；首末{ratio}"


def _format_anomaly_points(points, unit):
    formatted = []
    for point in (points or [])[:MAX_TEMPLATE_ANOMALY_POINTS]:
        timestamp = point.get("timestamp", "-")
        value = _format_value(point.get("value"), unit)
        ratio = _format_ratio(point.get("ratio"))
        kind = KIND_NAMES.get(point.get("kind"), point.get("kind", "峰值"))
        formatted.append(f"{timestamp}={value}({ratio},{kind})")
    return ",".join(formatted) if formatted else "无"


def _direction_point_label(summary):
    """方向未达到聚合阈值时，避免把代表性偏离值表述成已确认异常。"""
    return "异常值" if summary.get("anomalyPointStatus") == "threshold" else "代表偏离值"


def _format_pattern(value):
    return PATTERN_NAMES.get(value, value or "未知")


def _format_trend_name(value):
    if isinstance(value, dict):
        value = value.get("trend")
    return TREND_NAMES.get(value, value or "未知")


def main(data_json: str) -> dict:
    analysis_type = "file"
    data = json.loads(data_json or "{}")
    direction = DIRECTION_NAMES.get(analysis_type, analysis_type)
    status = STATUS_NAMES.get(data.get("status"), data.get("status", "未知"))
    active = data.get("activeThreadCount", 0)
    total = data.get("seriesCount", 0)
    candidate_count = data.get("candidateCount", 0)
    direction_signal_count = data.get("directionSignalCount", 0)
    contribution = float(data.get("candidateContribution", 0) or 0)
    max_score = float(data.get("maxScore", 0) or 0)
    unit = data.get("unit", "")
    lines = [
        f"### {direction}线程消耗候选",
        f"状态：{status}；活跃线程{active}/{total}；线程候选{candidate_count}；方向信号{direction_signal_count}；候选贡献{contribution:.1%}；最高评分{max_score:.2f}。",
    ]

    direction_summary = data.get("directionSummary") or {}
    if direction_summary:
        coverage = direction_summary.get("coverageRatio")
        coverage_text = "-" if coverage is None else f"{float(coverage):.0%}"
        lines.append(
            "- 方向趋势："
            + _format_trend(direction_summary, unit)
            + f"；模式{_format_pattern(direction_summary.get('pattern'))}"
            + f"；窗口{direction_summary.get('sampleCount', 0)}点"
            + f"({direction_summary.get('firstTimestamp', '-')}~{direction_summary.get('lastTimestamp', '-')})"
            + f"；基线{_format_value(direction_summary.get('baselineMedian'), unit)}"
            + f"；P95{_format_value(direction_summary.get('p95'), unit)}"
            + f"；峰值{_format_value(direction_summary.get('max'), unit)}@{direction_summary.get('peakTimestamp', '-')}"
            + f"；覆盖{coverage_text}"
            + f"；{_direction_point_label(direction_summary)}[{_format_anomaly_points(direction_summary.get('anomalyPoints'), unit)}]。"
        )
        if direction_summary.get("stepRatio") is not None:
            lines.append(
                "- 方向阶跃："
                f"{_format_value(direction_summary.get('preChangeMedian'), unit)}"
                f"->{_format_value(direction_summary.get('postChangeMedian'), unit)}"
                f"（{_format_ratio(direction_summary.get('stepRatio'))}）"
                f"@{direction_summary.get('stepTimestamp', '-')}。"
            )
        elif direction_summary.get("preChangeMedian") is not None:
            lines.append(
                "- 方向激活："
                f"{_format_value(direction_summary.get('preChangeMedian'), unit)}"
                f"->{_format_value(direction_summary.get('postChangeMedian'), unit)}"
                f"@{direction_summary.get('stepTimestamp', '-')}。"
            )

    for signal in data.get("directionSignals", []) or []:
        signal_type = TYPE_NAMES.get(signal.get("type"), signal.get("type", "方向信号"))
        signal_details = [
                signal_type,
                f"级别{LEVEL_NAMES.get(signal.get('level'), signal.get('level', '-'))}",
                f"评分{float(signal.get('score', 0)):.2f}",
                f"聚合中位{_format_value(signal.get('median'), unit)}",
                f"聚合峰值{_format_value(signal.get('peak'), unit)}",
                f"峰值倍率{_format_ratio(signal.get('peakRatio'))}",
                f"稳健Z值{_format_z(signal.get('robustZ'))}",
            f"峰值时刻{signal.get('peakTimestamp', '-')}",
        ]
        if signal.get("stepRatio") is not None:
            signal_details.append(
                f"阶跃倍率{_format_ratio(signal.get('stepRatio'))}"
                f"（{_format_value(signal.get('preChangeMedian'), unit)}->{_format_value(signal.get('postChangeMedian'), unit)}）"
            )
        elif signal.get("activationTimestamp") is not None:
            signal_details.append(
                f"激活变化{_format_value(signal.get('preChangeMedian'), unit)}"
                f"->{_format_value(signal.get('postChangeMedian'), unit)}"
                f"@{signal.get('activationTimestamp', '-')}"
            )
        if signal.get("type") == "collective_activation":
            signal_details.append(
                f"同步激活线程{signal.get('activationThreadCount', 0)}/{signal.get('activationThreadCoverage', 0)}"
            )
        elif signal.get("type") == "collective_step_change":
            signal_details.append(
                f"同步阶跃线程{signal.get('stepThreadCount', 0)}/{signal.get('stepThreadCoverage', 0)}"
            )
        else:
            signal_details.append(
                f"同步上涨线程{signal.get('risingThreadCount', 0)}/{signal.get('peakThreadCoverage', 0)}"
            )
        lines.append("- " + "；".join(signal_details))

    candidates = (data.get("candidates", []) or [])[:MAX_TEMPLATE_CANDIDATES]
    for item in candidates:
        anomaly_types = "/".join(TYPE_NAMES.get(value, value) for value in item.get("anomalyTypes", []))
        details = [
            f"线程{item.get('threadId', '-')}",
            f"评分{float(item.get('score', 0)):.2f}",
            f"模式{anomaly_types or _format_pattern(item.get('pattern'))}",
            f"中位{_format_value(item.get('median'), unit)}",
            f"峰值{_format_value(item.get('peak'), unit)}",
            f"峰值倍率{_format_ratio(item.get('peakRatio'))}",
            f"方向贡献{float(item.get('contribution', 0) or 0):.1%}",
            f"异常点{item.get('abnormalCount', 0)}/{item.get('sampleCount', 0)}",
        ]
        if item.get("trend"):
            details.append(f"趋势{_format_trend_name(item.get('trend'))}")
        details.append(f"异常值[{_format_anomaly_points(item.get('abnormalPoints'), unit)}]")
        if item.get("stepRatio") is not None:
            details.append(f"阶跃倍率{_format_ratio(item.get('stepRatio'))}")
        elif item.get("preChangeMedian") is not None:
            details.append(
                f"激活变化{_format_value(item.get('preChangeMedian'), unit)}"
                f"->{_format_value(item.get('postChangeMedian'), unit)}"
                f"@{item.get('stepTimestamp', '-')}"
            )
        if item.get("peakTimestamp"):
            details.append(f"峰值时刻{item['peakTimestamp']}")
        related = item.get("relatedEvidence", []) or []
        if related:
            evidence = ",".join(f"{entry['name']}(r={entry['correlation']:.2f})" for entry in related)
            details.append(f"辅助相关序列{evidence}")
        lines.append("- " + "；".join(details))

    if not candidates and not data.get("directionSignals"):
        direction_deviation = data.get("directionDeviation") or {}
        if direction_deviation.get("peakRatio", 1) > 1:
            lines.append(
                "- 未达阈值的方向聚合偏离："
                f"峰值倍率{_format_ratio(direction_deviation.get('peakRatio'))}；"
                f"稳健Z值{_format_z(direction_deviation.get('robustZ'))}。"
            )
        for item in (data.get("topDeviations", []) or [])[:MAX_TEMPLATE_DEVIATIONS]:
            lines.append(
                "- 未达阈值的线程偏离："
                f"线程{item.get('threadId', '-')}；"
                f"峰值倍率{_format_ratio(item.get('peakRatio'))}；"
                f"阶跃倍率{_format_ratio(item.get('stepRatio'))}；"
                f"稳健Z值{_format_z(item.get('robustZ'))}；"
                f"趋势{_format_trend_name(item.get('trend'))}；"
                f"弱偏离值[{_format_anomaly_points(item.get('abnormalPoints'), unit)}]；"
                f"方向贡献{float(item.get('contribution', 0) or 0):.1%}。"
            )

    omitted = max(0, int(data.get("returnedCandidateCount", len(candidates))) - len(candidates))
    omitted += int(data.get("truncatedCandidateCount", 0) or 0)
    if omitted:
        lines.append(f"- 另有{omitted}条低优先级候选未展开。")
    return {"result": "\n".join(lines)}
